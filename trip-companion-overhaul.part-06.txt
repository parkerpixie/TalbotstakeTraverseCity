  function bindHomeActions() {
    $('[data-new-fact]')?.addEventListener('click', () => {
      let next = state.factIndex;
      while (FUN_FACTS.length > 1 && next === state.factIndex) next = Math.floor(Math.random() * FUN_FACTS.length);
      state.factIndex = next;
      const target = $('#tcFunFact');
      if (target) target.textContent = FUN_FACTS[next];
    });
    $('[data-open-guide]')?.addEventListener('click', () => typeof showTab === 'function' ? showTab('discover') : showPanel('discover'));
    $('[data-car-game]')?.addEventListener('click', openCarGame);
    $('[data-get-location]')?.addEventListener('click', requestLocation);
    $('[data-house-address]')?.addEventListener('click', openHouseAddressDialog);
    $('[data-open-house]')?.addEventListener('click', () => typeof showTab === 'function' ? showTab('house') : showPanel('house'));
    $$('[data-panel="home"] [data-tc-tab]').forEach(button => button.addEventListener('click', () => showPanel(button.dataset.tcTab)));
  }

  function ensureDialogs() {
    if (!$('#tcCarGameDialog')) {
      const dialog = document.createElement('dialog');
      dialog.id = 'tcCarGameDialog';
      dialog.className = 'tc-overhaul-dialog';
      dialog.innerHTML = '<button type="button" class="tc-dialog-close" data-close-dialog>×</button><div id="tcCarGameContent"></div>';
      document.body.appendChild(dialog);
      dialog.querySelector('[data-close-dialog]').addEventListener('click', () => dialog.close());
    }
    if (!$('#tcHouseAddressDialog')) {
      const dialog = document.createElement('dialog');
      dialog.id = 'tcHouseAddressDialog';
      dialog.className = 'tc-overhaul-dialog';
      dialog.innerHTML = `<button type="button" class="tc-dialog-close" data-close-dialog>×</button><p class="eyebrow dark">Private house address</p><h2>Keep the rental out of public GitHub.</h2><p>This address is stored only in this browser on this device. It is used to build the Google Maps directions button.</p><form method="dialog" id="tcHouseAddressForm"><label>Street address<input id="tcHouseAddressInput" autocomplete="street-address" placeholder="Paste the Traverse City rental address"></label><div class="tc-dialog-actions"><button type="submit" value="cancel">Cancel</button><button type="button" class="tc-primary-button" data-save-house-address>Save privately</button></div></form>`;
      document.body.appendChild(dialog);
      dialog.querySelector('[data-close-dialog]').addEventListener('click', () => dialog.close());
      dialog.querySelector('[data-save-house-address]').addEventListener('click', () => {
        const value = $('#tcHouseAddressInput')?.value.trim();
        if (value) localStorage.setItem(HOUSE_ADDRESS_KEY, value);
        else localStorage.removeItem(HOUSE_ADDRESS_KEY);
        dialog.close();
        renderHome();
      });
    }
  }

  function openCarGame() {
    ensureDialogs();
    const game = CAR_GAMES[Math.floor(Math.random() * CAR_GAMES.length)];
    $('#tcCarGameContent').innerHTML = `<div class="tc-game-icon">${game.icon}</div><p class="eyebrow dark">Road game generator</p><h2>${safe(game.name)}</h2><p>${safe(game.text)}</p><button type="button" class="tc-primary-button" data-another-game>Give us another</button>`;
    $('#tcCarGameContent [data-another-game]').addEventListener('click', openCarGame);
    const dialog = $('#tcCarGameDialog');
    if (!dialog.open) dialog.showModal();
  }

  function openHouseAddressDialog() {
    ensureDialogs();
    $('#tcHouseAddressInput').value = localStorage.getItem(HOUSE_ADDRESS_KEY) || '';
    $('#tcHouseAddressDialog').showModal();
  }
